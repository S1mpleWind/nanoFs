cmake_minimum_required(VERSION 3.20)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
project(cse_2026_lab2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(NANOFS_STAGE_B "Build Lab1B remote filesystem" ON)
option(NANOFS_LAB2B "Enable Lab2B locking and concurrency" OFF)
option(NANOFS_REQUIRE_FUSE "Fail configuration if FUSE is unavailable" OFF)
option(NANORAFT_LAB3 "Build Lab 3 Raft/KV components and tests" ON)
enable_testing()
add_subdirectory(third_party/googletest)
if(NANOFS_STAGE_B)
  add_subdirectory(third_party/rpclib)
  add_subdirectory(librpc)
endif()
add_subdirectory(nanofs)
if(NANORAFT_LAB3)
  if(NOT NANOFS_STAGE_B)
    message(FATAL_ERROR "NANORAFT_LAB3 requires NANOFS_STAGE_B=ON for librpc")
  endif()
  add_subdirectory(nanoraft)
endif()
